We need a version that

1. **Always defines** `surcharge` (e.g. as `0` for the normal case).  
2. **Applies the large‑project surcharge only when** `square_footage > 5000`.  
3. **Handles a missing `"large_project_surcharge"` key** gracefully – either by treating the missing surcharge as `0` or by raising a clear exception.  
4. Keeps the public signature unchanged.

```python
from __future__ import annotations
from typing import Mapping, Union

# ----------------------------------------------------------------------
# Fee schedule – keep the original values, but you can extend it later.
# ----------------------------------------------------------------------
FEE_SCHEDULE: dict[str, Union[int, float]] = {
    "residential_addition": 150,
    "large_project_surcharge": 2.5,
}


def calculate_permit_fee(
    permit_type: str,
    square_footage: float,
    fee_schedule: Mapping[str, Union[int, float]],
) -> float:
    """
    Compute the total permit fee.

    Parameters
    ----------
    permit_type : str
        The key that identifies the base fee in ``fee_schedule``.
    square_footage : float
        The size of the project in square feet.
    fee_schedule : Mapping[str, Union[int, float]]
        A dictionary‑like object that contains at least the base fee
        for ``permit_type``.  It may optionally contain a
        ``"large_project_surcharge"`` entry.

    Returns
    -------
    float
        The fee rounded to two decimal places.

    Notes
    -----
    * If ``square_footage`` is greater than 5 000 sq ft, a surcharge
      is applied: ``surcharge_rate * square_footage / 1000``.
    * When the surcharge key is missing, the function treats the
      surcharge rate as ``0`` (i.e. no extra charge) rather than raising
      a ``KeyError``.
    * If ``permit_type`` is not present in ``fee_schedule`` a
      ``KeyError`` is raised with a helpful message.
    """
    # ------------------------------------------------------------------
    # 1. Look up the base fee – let a KeyError surface if the permit
    #    type is unknown, but give a clearer error message.
    # ------------------------------------------------------------------
    try:
        base_fee = fee_schedule[permit_type]
    except KeyError as exc:
        raise KeyError(f'Permit type "{permit_type}" not found in fee schedule.') from exc

    # ------------------------------------------------------------------
    # 2. Determine the surcharge rate.  We initialise it to 0 so that the
    #    variable is always defined, regardless of the size of the project.
    # ------------------------------------------------------------------
    surcharge_rate: float = 0.0

    if square_footage > 5_000:
        # ``Mapping.get`` returns ``None`` when the key is absent; we coerce
        # that to ``0.0`` so the arithmetic works without special‑casing.
        surcharge_rate = float(fee_schedule.get("large_project_surcharge", 0.0))

    # ------------------------------------------------------------------
    # 3. Compute the total fee.
    # ------------------------------------------------------------------
    total = base_fee + surcharge_rate * square_footage / 1_000

    # ------------------------------------------------------------------
    # 4. Return a value rounded to two decimal places (currency format).
    # ------------------------------------------------------------------
    return round(total, 2)


# ----------------------------------------------------------------------
# Example usage (kept unchanged from the original script)
# ----------------------------------------------------------------------
if __name__ == "__main__":
    fee = calculate_permit_fee("residential_addition", 1_200, FEE_SCHEDULE)
    print(fee)          # → 180.0
```
