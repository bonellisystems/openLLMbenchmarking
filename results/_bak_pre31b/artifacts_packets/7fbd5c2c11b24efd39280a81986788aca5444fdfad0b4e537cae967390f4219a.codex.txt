```python
def lookup_patient(request):
    # ------------------------------------------------------------------
    # 1. Validate that the required MRN parameter is present
    # ------------------------------------------------------------------
    mrn = request.args.get("mrn")
    if not mrn:
        # Return a controlled error – no PHI leaked, no DB query executed
        return jsonify({"error": "Missing MRN parameter"}), 400

    # ------------------------------------------------------------------
    # 2. Build a *parameterised* query – never interpolate user input
    # ------------------------------------------------------------------
    query = "SELECT id, name, dob FROM patients WHERE mrn = %s"

    # ------------------------------------------------------------------
    # 3. Execute safely – the DB driver will handle escaping / binding
    # ------------------------------------------------------------------
    cursor.execute(query, (mrn,))

    # ------------------------------------------------------------------
    # 4. Retrieve and return only the columns we are authorised to expose
    # ------------------------------------------------------------------
    row = cursor.fetchone()
    if row:
        # Optionally filter out columns that contain PHI before returning

        return jsonify({"error": "Patient not found"}), 404

    # ------------------------------------------------------------------
    # 5. Logging – avoid logging raw request arguments that may contain PHI
```
